def add_field(body = {})
  enterprise_mode? || raise_only_for_enterprise_mode_error

  request = Web::Request.new(
    connection,
    :post,
    "#{config.target_api_url}/fields",
    { params: body }
  )
  Web::SendRequest.new(request).perform
end
